# Sidecar runtime. Pinned exactly: a silent minor bump changes which kernels the
# NPU plugin emits, and the device measurements in the PRD stop meaning anything.
#
# 2026.3.0 is the version the target machine was profiled on (see the PRD, §9).
# Moving to 2026.3.1 is a deliberate change with its own re-measurement, not a
# drive-by upgrade.
openvino==2026.3.0
openvino-genai==2026.3.0.0

# Torch, for Russian speech, and the rule it breaks.
#
# The sidecar was meant to keep torch out of the always-on process. That rule
# was written to keep the process lean, and the process is not lean any more:
# RISK-9 measured the idle-unload policy at twelve seconds to the first spoken
# reply, so both models are now resident and the sidecar holds ~3.4 GB by
# design. Torch adds 183 MB and Silero 99 more, peaking near 520 MB after a
# synthesis — 15% on top of what is already there.
#
# The alternative was converting Silero to OpenVINO IR, and it does not exist:
# the model is one TorchScript system whose forward takes strings, with accent
# placement baked into the graph. OpenVINO refuses it on SequenceInsert, and
# there is no tensors-in-tensors-out boundary to export instead.
#
# CPU build on purpose: synthesis runs on the CPU so it never contends with the
# iGPU that recognition and the narrator share.
torch==2.14.0
omegaconf==2.3.0

# Hearing.
#
# PyAudioWPatch rather than sounddevice, and the reason is RISK-3 rather than
# taste: measuring the application's own speech in the microphone signal needs
# WASAPI *loopback* capture, and sounddevice 0.5.6 exposes exclusive,
# auto_convert and explicit_sample_format — no loopback at all. Verified on this
# machine: PyAudioWPatch enumerates both the microphone and the speakers as
# capture devices; sounddevice sees only the microphone. Discovering that after
# building the capture layer would have meant rewriting it.
PyAudioWPatch==0.2.12.8

# Voice activity detection, ONNX build: no torchaudio, shares the onnxruntime the
# wake word needs anyway, and measured at 0.27 ms a frame against the 1 ms the
# design budgeted.
silero-vad==6.2.1
onnxruntime==1.29.0
